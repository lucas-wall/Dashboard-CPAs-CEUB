/*
  Medidas DAX para o Power BI
  Base de Dados: CEUB_export-data-desafio-pbi.xlsx

  Objetivo: Criar medidas que permitam análises aprofundadas e a construção de dashboards interativos,
  aproveitando as transformações realizadas em M Language.
*/

-- 1. Medidas Básicas

// Total de Avaliações
Total Avaliacoes = COUNTROWS('CPA - CEUB')

// Média da Nota Normalizada
Media Nota Normalizada = AVERAGE('CPA - CEUB'[Nota_Normalizada])

// Total de Respostas Descritivas (para identificar o engajamento com questões abertas)
Total Respostas Descritivas = CALCULATE(COUNTROWS('CPA - CEUB'), NOT ISBLANK('CPA - CEUB'[Resposta_Descritiva]))

-- 2. Medidas de Análise Temporal

// Média da Nota Normalizada por Mês (para tendências)
Media Nota Normalizada Mes = 
CALCULATE(
    AVERAGE('CPA - CEUB'[Nota_Normalizada]),
    DATESMTD('CPA - CEUB'[Data_Avaliacao])
)

// Média da Nota Normalizada Ano a Ano (para comparação anual)
Media Nota Normalizada Ano = 
CALCULATE(
    AVERAGE('CPA - CEUB'[Nota_Normalizada]),
    DATESYTD('CPA - CEUB'[Data_Avaliacao])
)

-- 3. Medidas de Análise de Desempenho (Exemplo: Avaliação por Questão/Grupo)

// Média da Nota por Questão
Media Nota por Questao = 
CALCULATE(
    AVERAGE('CPA - CEUB'[Nota_Normalizada]),
    ALLEXCEPT('CPA - CEUB', 'CPA - CEUB'[Questão])
)

// Média da Nota por Tipo de Avaliação
Media Nota por Tipo Avaliacao = 
CALCULATE(
    AVERAGE('CPA - CEUB'[Nota_Normalizada]),
    ALLEXCEPT('CPA - CEUB', 'CPA - CEUB'[Tipo_Avaliacao])
)

// Média da Nota por Professor
Media Nota por Professor = 
CALCULATE(
    AVERAGE('CPA - CEUB'[Nota_Normalizada]),
    ALLEXCEPT('CPA - CEUB', 'CPA - CEUB'[Professor])
)

-- 4. Medidas de Classificação/Segmentação (Exemplo: Contagem de Respostas por Categoria)

// Contagem de Respostas 'Concordo Totalmente'
Contagem Concordo Totalmente = 
CALCULATE(
    COUNTROWS('CPA - CEUB'),
    'CPA - CEUB'[Resposta] = "a) Concordo totalmente"
)

// Contagem de Respostas 'Discordo Totalmente'
Contagem Discordo Totalmente = 
CALCULATE(
    COUNTROWS('CPA - CEUB'),
    'CPA - CEUB'[Resposta] = "e) Discordo totalmente"
)

-- 5. Medidas de Comparação (Exemplo: Variação em relação ao período anterior)

// Média da Nota Normalizada - Mês Anterior
Media Nota Normalizada Mes Anterior = 
CALCULATE(
    AVERAGE('CPA - CEUB'[Nota_Normalizada]),
    PREVIOUSMONTH('CPA - CEUB'[Data_Avaliacao])
)

// Variação da Média da Nota Normalizada em relação ao Mês Anterior
Variacao Media Nota Normalizada Mes Anterior = 
[Media Nota Normalizada] - [Media Nota Normalizada Mes Anterior]

-- 6. Medidas para Análise de Alunos/Cursos (Exemplo: Contagem de Cursos EAD)

// Total de Cursos EAD
Total Cursos EAD = 
CALCULATE(
    DISTINCTCOUNT('CPA - CEUB'[Curso]),
    'CPA - CEUB'[Curso_EAD] = TRUE()
)

// Total de Avaliações por Campus
Total Avaliacoes por Campus = 
CALCULATE(
    COUNTROWS('CPA - CEUB'),
    ALLEXCEPT('CPA - CEUB', 'CPA - CEUB'[Nome_Campus])
)

-- 7. Medidas para Análise de Pergunta Específica (Exemplo: Média da Nota para uma Questão específica)

// Média da Nota para Questão 1 (Exemplo)
Media Nota Questao 1 = 
CALCULATE(
    AVERAGE('CPA - CEUB'[Nota_Normalizada]),
