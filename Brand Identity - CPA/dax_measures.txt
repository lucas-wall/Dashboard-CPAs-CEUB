/*
  DAX measures for Power BI
  Data source: CEUB_export-data-challenge-pbi.xlsx

  Goal: build measures that enable in-depth analysis and interactive dashboards,
  on top of the transformations done in M (Power Query).
  Note: column names and answer values (e.g. 'a) Concordo totalmente') are kept
  exactly as they appear in the source data model.
*/

-- 1. Basic measures

// Total number of evaluations
Total Evaluations = COUNTROWS('CPA - CEUB')

// Average normalized score
Avg Normalized Score = AVERAGE('CPA - CEUB'[Nota_Normalizada])

// Total open-ended answers (engagement with open questions)
Total Open-Ended Answers = CALCULATE(COUNTROWS('CPA - CEUB'), NOT ISBLANK('CPA - CEUB'[Resposta_Descritiva]))

-- 2. Time-based measures

// Average normalized score, month to date (trends)
Avg Normalized Score MTD = 
CALCULATE(
    AVERAGE('CPA - CEUB'[Nota_Normalizada]),
    DATESMTD('CPA - CEUB'[Data_Avaliacao])
)

// Average normalized score, year to date (annual comparison)
Avg Normalized Score YTD = 
CALCULATE(
    AVERAGE('CPA - CEUB'[Nota_Normalizada]),
    DATESYTD('CPA - CEUB'[Data_Avaliacao])
)

-- 3. Performance measures (e.g. score by question/group)

// Average score by Question
Avg Score by Question = 
CALCULATE(
    AVERAGE('CPA - CEUB'[Nota_Normalizada]),
    ALLEXCEPT('CPA - CEUB', 'CPA - CEUB'[Questão])
)

// Average score by Evaluation Type
Avg Score by Evaluation Type = 
CALCULATE(
    AVERAGE('CPA - CEUB'[Nota_Normalizada]),
    ALLEXCEPT('CPA - CEUB', 'CPA - CEUB'[Tipo_Avaliacao])
)

// Average score by Professor
Avg Score by Professor = 
CALCULATE(
    AVERAGE('CPA - CEUB'[Nota_Normalizada]),
    ALLEXCEPT('CPA - CEUB', 'CPA - CEUB'[Professor])
)

-- 4. Classification measures (e.g. answer count by category)

// Count of 'Strongly agree' answers
Count Strongly Agree = 
CALCULATE(
    COUNTROWS('CPA - CEUB'),
    'CPA - CEUB'[Resposta] = "a) Concordo totalmente"
)

// Count of 'Strongly disagree' answers
Count Strongly Disagree = 
CALCULATE(
    COUNTROWS('CPA - CEUB'),
    'CPA - CEUB'[Resposta] = "e) Discordo totalmente"
)

-- 5. Comparison measures (e.g. change vs previous period)

// Average normalized score - previous month
Avg Normalized Score Prev Month = 
CALCULATE(
    AVERAGE('CPA - CEUB'[Nota_Normalizada]),
    PREVIOUSMONTH('CPA - CEUB'[Data_Avaliacao])
)

// Change in average normalized score vs previous month
Avg Normalized Score MoM Change = 
[Avg Normalized Score] - [Avg Normalized Score Prev Month]

-- 6. Student/program measures (e.g. distance learning program count)

// Total distance learning (EAD) programs
Total EAD Programs = 
CALCULATE(
    DISTINCTCOUNT('CPA - CEUB'[Curso]),
    'CPA - CEUB'[Curso_EAD] = TRUE()
)

// Total evaluations by campus
Total Evaluations by Campus = 
CALCULATE(
    COUNTROWS('CPA - CEUB'),
    ALLEXCEPT('CPA - CEUB', 'CPA - CEUB'[Nome_Campus])
)

-- 7. Single-question measures (e.g. average score for one specific question)

// Average score for question 1 (example)
Avg Score Question 1 = 
CALCULATE(
    AVERAGE('CPA - CEUB'[Nota_Normalizada]),
